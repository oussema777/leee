# Copy this entire block into the existing VPS PuTTY session.
# Changes only the dedicated LEE gateway site, then validates and reloads Nginx.
(
set -eu
FILE=/etc/nginx/sites-available/lee-aramex-gateway.conf
SITE=/etc/nginx/sites-enabled/lee-aramex-gateway.conf
if [ "$(readlink -f "$SITE")" != "$FILE" ]; then
    echo "Unexpected gateway site link. Stop here and send the output."
    exit 1
fi
systemctl is-active --quiet lee-aramex-gateway
curl --fail --silent --show-error --max-time 5 http://127.0.0.1:5088/health
printf '\n'
BACKUP=$(mktemp /etc/nginx/sites-available/lee-aramex-backup.XXXXXX)
cp "$FILE" "$BACKUP"
cat > "$FILE" <<'LEE_NGINX'
# These named limits belong only to the LEE gateway.
limit_req_zone $server_name zone=lee_aramex_requests:1m rate=30r/m;
limit_conn_zone $server_name zone=lee_aramex_connections:1m;

server {
    listen 80;
    server_name aramex-gateway.theleeexperience.com;
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/lee-aramex-acme;
    }
    location / { return 503; }
}

server {
    listen 443 ssl;
    server_name aramex-gateway.theleeexperience.com;
    ssl_certificate /etc/letsencrypt/live/lee-aramex-gateway/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/lee-aramex-gateway/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;

    access_log off;
    client_max_body_size 256k;
    client_body_buffer_size 256k;
    client_body_timeout 10s;
    send_timeout 20s;
    keepalive_timeout 10s;
    limit_req zone=lee_aramex_requests burst=5 nodelay;
    limit_req_status 429;
    limit_conn lee_aramex_connections 4;
    limit_conn_status 429;

    proxy_http_version 1.1;
    proxy_pass_request_headers off;
    proxy_set_header Host localhost;
    proxy_set_header Connection close;
    proxy_set_header Authorization $http_authorization;
    proxy_set_header Content-Type $http_content_type;
    proxy_connect_timeout 3s;
    proxy_send_timeout 10s;
    proxy_read_timeout 20s;
    proxy_next_upstream off;
    proxy_request_buffering on;
    proxy_buffering on;
    proxy_max_temp_file_size 0;
    proxy_cache off;
    proxy_redirect off;
    add_header Cache-Control "no-store" always;

    if ($args != "") { return 404; }

    location = /health {
        if ($request_method != GET) { return 405; }
        proxy_pass http://127.0.0.1:5088;
    }
    location = /uat/shipments {
        if ($request_method != POST) { return 405; }
        proxy_pass http://127.0.0.1:5088;
    }
    location = /uat/tracking {
        if ($request_method != POST) { return 405; }
        proxy_pass http://127.0.0.1:5088;
    }
    location = /uat/label {
        if ($request_method != POST) { return 405; }
        proxy_pass http://127.0.0.1:5088;
    }
    location / { return 404; }
}
LEE_NGINX

if nginx -t && systemctl reload nginx; then
    echo "Gateway HTTPS routing enabled. Backup: $BACKUP"
else
    cp "$BACKUP" "$FILE"
    nginx -t && systemctl reload nginx
    echo "Gateway configuration restored. Stop here and send the output."
    exit 1
fi

# These checks send no authorized request to Aramex.
sleep 2
HEALTH=$(curl --silent --show-error --max-time 10 --resolve aramex-gateway.theleeexperience.com:443:127.0.0.1 -o /dev/null -w '%{http_code}' https://aramex-gateway.theleeexperience.com/health) || HEALTH=000
AUTH=$(curl --silent --show-error --max-time 10 --resolve aramex-gateway.theleeexperience.com:443:127.0.0.1 -o /dev/null -w '%{http_code}' -H 'Content-Type: application/json' -d '{}' https://aramex-gateway.theleeexperience.com/uat/tracking) || AUTH=000
if [ "$HEALTH" != "200" ] || [ "$AUTH" != "401" ]; then
    cp "$BACKUP" "$FILE"
    nginx -t && systemctl reload nginx
    echo "Checks failed: health=$HEALTH authentication=$AUTH. Gateway configuration restored."
    exit 1
fi
echo "PASS: HTTPS health=200; missing gateway key=401. No Aramex request sent."
echo "Eventra process status:"
pm2 list
)
